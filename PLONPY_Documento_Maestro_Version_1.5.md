# PARTE 1: ECONOMÍA DE CHISPA ✦

## Filosofía Económica

> **"CHISPA compra comodidad, no progreso. Los Núcleos se encuentran, no se compran."**

La economía de PLONPY se rige por tres principios:

1. **El jugador free-to-play debe poder acceder a todo eventualmente.** Sin muros de pago.
2. **El jugador que paga obtiene conveniencia y cosméticos, no ventajas decisivas.**
3. **La CHISPA no debe sentirse como un obstáculo, sino como una recompensa adicional.**

## 3.2 SCRIPT: [LLAMADA.GD](https://llamada.gd/)

gdscript

```
extends Node3D

# Sistema de Llamada
# Guía al jugador sin brújula ni flechas

@export var objetivo: Node3D  # El Núcleo a encontrar
@export var distancia_maxima: float = 100.0  # Distancia máxima de detección
@export var distancia_minima: float = 5.0    # Distancia de descubrimiento

@onready var audio_pulso: AudioStreamPlayer3D = $AudioPulso
@onready var plonpy: Node3D = get_tree().get_first_node_in_group("plonpy")

var intensidad: float = 0.0
var direccion: Vector3 = Vector3.ZERO

func _ready():
    audio_pulso.stream = preload("res://assets/audio/pulso_llamada.ogg")
    audio_pulso.play()

func _process(delta):
    if not objetivo:
        return
    
    # Calcular distancia
    var distancia = plonpy.global_position.distance_to(objetivo.global_position)
    
    # Calcular intensidad (1 = cerca, 0 = lejos)
    intensidad = clamp(1.0 - (distancia / distancia_maxima), 0.0, 1.0)
    
    # Calcular dirección
    direccion = (objetivo.global_position - plonpy.global_position).normalized()
    
    # Actualizar audio
    audio_pulso.volume_db = lerp(-40.0, 0.0, intensidad)
    audio_pulso.pitch_scale = lerp(0.8, 1.2, intensidad)
    
    # Actualizar Plonpy
    plonpy.actualizar_llamada(intensidad, direccion, delta)
    
    # Detectar descubrimiento
    if distancia <= distancia_minima:
        descubrir_nucleo()

func descubrir_nucleo():
    if not objetivo:
        return
    
    # Reproducir sonido de descubrimiento
    var audio = AudioStreamPlayer.new()
    audio.stream = preload("res://assets/audio/descubrimiento.ogg")
    add_child(audio)
    audio.play()
    
    # Notificar al GameManager
    GameManager.descubrir_nucleo(objetivo)
    
    # Eliminar el Núcleo del mundo
    objetivo.queue_free()
    objetivo = null
```

## 3.3 SCRIPT: [PLONPY.GD](https://plonpy.gd/)

gdscript

```
extends CharacterBody3D

# Plonpy base
# Se mueve, reacciona a La Llamada, evoluciona

@export var velocidad: float = 5.0
@export var velocidad_rotacion: float = 3.0

@onready var modelo: Node3D = $Modelo
@onready var nucleo_interno: MeshInstance3D = $Modelo/NucleoInterno
@onready var ojos: Node3D = $Modelo/Ojos

var intensidad_llamada: float = 0.0
var direccion_llamada: Vector3 = Vector3.ZERO
var estado: String = "base"  # base, sensor, observador, vidente, primigenio

func _physics_process(delta):
    # Movimiento básico
    var input_dir = Input.get_vector("ui_left", "ui_right", "ui_up", "ui_down")
    var direction = (transform.basis * Vector3(input_dir.x, 0, input_dir.y)).normalized()
    
    if direction:
        velocity.x = direction.x * velocidad
        velocity.z = direction.z * velocidad
    else:
        velocity.x = move_toward(velocity.x, 0, velocidad)
        velocity.z = move_toward(velocity.z, 0, velocidad)
    
    move_and_slide()
    
    # Reaccionar a La Llamada
    if intensidad_llamada > 0.1:
        reaccionar_llamada(delta)

func reaccionar_llamada(delta):
    # 1. Brillo corporal
    var material = modelo.get_active_material(0)
    if material:
        material.emission_energy = lerp(0.0, 2.0, intensidad_llamada)
    
    # 2. Pulso del núcleo interno
    var escala = 1.0 + sin(Time.get_ticks_msec() * 0.005) * 0.2 * intensidad_llamada
    nucleo_interno.scale = Vector3(escala, escala, escala)
    
    # 3. Rotación sutil hacia la dirección
    if direccion_llamada.length() > 0:
        var target_rotation = atan2(direccion_llamada.x, direccion_llamada.z)
        var current_rotation = rotation.y
        rotation.y = lerp_angle(current_rotation, target_rotation, delta * velocidad_rotacion * intensidad_llamada)
    
    # 4. Ojos cambian de color
    for ojo in ojos.get_children():
        var mat = ojo.get_active_material(0)
        if mat:
            mat.albedo_color = Color(0.5, 0.8, 1.0).lerp(Color(1.0, 1.0, 0.5), intensidad_llamada)

func actualizar_llamada(intensidad: float, direccion: Vector3, delta: float):
    intensidad_llamada = intensidad
    direccion_llamada = direccion

func evolucionar(nuevo_estado: String):
    estado = nuevo_estado
    
    # Cambiar modelo
    var nuevo_modelo = load("res://assets/models/plonpy_" + nuevo_estado + ".glb")
    if nuevo_modelo:
        modelo.mesh = nuevo_modelo
    
    # Actualizar estadísticas
    match nuevo_estado:
        "sensor":
            velocidad = 6.0
        "observador":
            velocidad = 7.0
        "vidente":
            velocidad = 8.0
        "primigenio":
            velocidad = 10.0
    
    # Efecto visual de evolución
    var tween = create_tween()
    tween.tween_property(modelo, "scale", Vector3(1.2, 1.2, 1.2), 0.5)
    tween.tween_property(modelo, "scale", Vector3(1.0, 1.0, 1.0), 0.5)
```

var chispa: int = 0

    "comun": 10,
    "raro": 50,
    "epico": 250,
    "legendario": 2000
}

const XP_POR_NIVEL = {
    1: 0, 2: 100, 3: 250, 4: 500, 5: 900,
    6: 1500, 7: 2300, 8: 3400, 9: 4800, 10: 6500
}

func descubrir_nucleo(nucleo: Node3D):
    var id = nucleo.nucleo_id
    var rareza = nucleo.rareza
    
    # Añadir a colección
    nucleos_descubiertos.append(id)
    
    # Dar XP
    var xp = XP_POR_NUCLEO[rareza]
    ganar_xp(xp)
    
    # Dar CHISPA
    var chispa_ganada = 5 if rareza == "comun" else 25 if rareza == "raro" else 100
    chispa += chispa_ganada
    
    # Emitir señal
    nucleo_descubierto.emit(nucleo)
    
    # Verificar evolución
    verificar_evolucion()

func ganar_xp(cantidad: int):
    xp_total += cantidad
    xp_ganada.emit(cantidad)
    
    # Verificar subida de nivel
    var nuevo_nivel = nivel
    for n in XP_POR_NIVEL:
        if xp_total >= XP_POR_NIVEL[n]:
            nuevo_nivel = n
    
    if nuevo_nivel > nivel:
        nivel = nuevo_nivel
        nivel_subido.emit(nivel)

    if nivel >= 5 and "001" in nucleos_descubiertos and "002" in nucleos_descubiertos and "003" in nucleos_descubiertos:
        if not "zyl_sensor" in nucleos_descubiertos:
            evolucion_disponible.emit("zyl")
